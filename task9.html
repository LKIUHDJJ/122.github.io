Task9 第九个帖子散乱的密文

解题步骤

使用 Code::Blocks 运行C语言解密代码

#include <stdio.h>
#include <string.h>
int main()
{
    // 密文和密钥
    char cipher[] = "If5{ag024c483549d7fd@@@1";
    char key[] = "216534";
    int n = strlen(key); // n=6
    char result[1000] = {0};
    int res_idx = 0;
    // 分组，每6个字符一组
    for(int i = 0; i < strlen(cipher); i += n)
    {
        char block[7] = {0};
        char temp[7] = {0};
        strncpy(block, cipher+i, n); // 取出一组6字符
        // 核心置换逻辑，和图片Python代码一模一样
        for(int j = 0; j < n; j++)
        {
            int pos = key[j] - '0' - 1; // key数字转下标，-1
            temp[pos] = block[j];
        }
        // 把本组解密结果拼到总结果
        strcat(result, temp);
    }
    printf("解密完整结果：%s\n",result);
    // 删除末尾填充的@符号
    char final_flag[1000]={0};
    int p=0;
    for(int i=0;result[i]!='\0';i++){
        if(result[i]!='@'){
            final_flag[p++] = result[i];
        }
    }
    final_flag[p] = '\0';
    printf("去掉填充符后的flag：%s\n",final_flag);
    return 0;
}

操作说明：

1、打开Code::Blocks，新建空白C项目，把全部代码粘贴进去。

2、编译运行，程序会输出两段内容：完整解密字符串，以及去除填充字符@之后的最终flag。

原理：分组换位密码，按密钥数字做字符位置置换；@为填充字符，解密完成需要剔除。
